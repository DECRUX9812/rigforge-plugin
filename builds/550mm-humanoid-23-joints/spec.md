/* 550mm humanoid · 23 joints — ESP32-S3 DevKitC-1
 * Generated by Rigforge. 23 serial-bus joints on two TTL buses.
 *
 * THIS IS A STAND-AND-HOLD CONTROLLER. It does not walk. A biped cannot be made
 * stable by geometry — it needs a balance loop reading the IMU at 200Hz and
 * commanding hips/ankles every cycle. That is the next project, and this file is
 * the honest starting point for it.
 */
#include <SCServo.h>
#include <Wire.h>

SMS_STS st;                       // legs: bus 0
SMS_STS upper;                    // waist + arms + head: bus 1

// Servo IDs are per-bus, so the same ID can exist on each bus:
// legs  ID  1: Left hip yaw
// legs  ID  2: Left hip roll
// legs  ID  3: Left hip pitch
// legs  ID  4: Left knee
// legs  ID  5: Left ankle pitch
// legs  ID  6: Left ankle roll
// legs  ID  7: Right hip yaw
// legs  ID  8: Right hip roll
// legs  ID  9: Right hip pitch
// legs  ID 10: Right knee
// legs  ID 11: Right ankle pitch
// legs  ID 12: Right ankle roll
// upper ID  1: Waist yaw
// upper ID  2: Head pan
// upper ID  3: Head tilt
// upper ID  4: Left shoulder roll
// upper ID  5: Left shoulder pitch
// upper ID  6: Left elbow
// upper ID  7: Left wrist
// upper ID  8: Right shoulder roll
// upper ID  9: Right shoulder pitch
// upper ID 10: Right elbow
// upper ID 11: Right wrist
const long BAUDRATE = 1000000;

void setupBuses() {
  // UART1 = legs, UART2 = arms + head. UART0 stays the console.
  Serial1.begin(BAUDRATE, SERIAL_8N1, 18, 17);
  st.pSerial = &Serial1;
  Serial2.begin(BAUDRATE, SERIAL_8N1, 48, 47);
  upper.pSerial = &Serial2;
}

void standPose() {
  // Every joint to its centre with a slow, low-torque-limit move: on first power
  // up you do NOT want 23 servos snapping to position with the robot upright.
  st.WritePosEx(1, 2048, 120, 30);
  st.WritePosEx(2, 2048, 120, 30);
  st.WritePosEx(3, 2048, 120, 30);
  st.WritePosEx(4, 2048, 120, 30);
  st.WritePosEx(5, 2048, 120, 30);
  st.WritePosEx(6, 2048, 120, 30);
  st.WritePosEx(7, 2048, 120, 30);
  st.WritePosEx(8, 2048, 120, 30);
  st.WritePosEx(9, 2048, 120, 30);
  st.WritePosEx(10, 2048, 120, 30);
  st.WritePosEx(11, 2048, 120, 30);
  st.WritePosEx(12, 2048, 120, 30);
  upper.WritePosEx(1, 2048, 120, 30);
  upper.WritePosEx(2, 2048, 120, 30);
  upper.WritePosEx(3, 2048, 120, 30);
  upper.WritePosEx(4, 2048, 120, 30);
  upper.WritePosEx(5, 2048, 120, 30);
  upper.WritePosEx(6, 2048, 120, 30);
  upper.WritePosEx(7, 2048, 120, 30);
  upper.WritePosEx(8, 2048, 120, 30);
  upper.WritePosEx(9, 2048, 120, 30);
  upper.WritePosEx(10, 2048, 120, 30);
  upper.WritePosEx(11, 2048, 120, 30);
  delay(1200);
}

void holdStand() {
  // Torque on, position held. A real controller replaces this with a balance loop.
  for (int id = 1; id <= 12; id++) st.EnableTorque(id, 1);
  for (int id = 1; id <= 11; id++) upper.EnableTorque(id, 1);
}

void report() {
  // A joint pinned at high load is a servo about to strip a gear.
  for (int id = 1; id <= 12; id++)
    Serial.printf("leg %02d pos=%4d load=%4d\n", id, st.ReadPos(id), st.ReadLoad(id));
  for (int id = 1; id <= 11; id++)
    Serial.printf("up  %02d pos=%4d load=%4d\n", id, upper.ReadPos(id), upper.ReadLoad(id));
}

void setup() {
  Serial.begin(115200);
  setupBuses();
  standPose();
  holdStand();
  Serial.println("550mm humanoid · 23 joints: standing. Balance loop not implemented — see the header.");
}

void loop() {
  // No gait. Read the IMU, watch the joint loads, and stop if anything is
  // fighting: a joint pinned at high load is a servo about to strip a gear.
  static uint32_t last = 0;
  if (millis() - last > 250) {
    last = millis();
    report();
  }
}

## Assembly

1. **Print the frame** (38.2 h) — 18 printed parts, 873g PETG, 38h on a 0.4mm nozzle at 0.2mm layers. Orient load-bearing links so layer lines run along the load, not across it. _Tools: printer, slicer._
2. **Print the skeleton in PETG** (38.2 h) — 873g over 38h. Thighs, shins and hip brackets take repeated shock loads — PETG, 4 walls, and print them upright so the layer lines run along the limb, not across it. _Tools: printer, slicer._
3. **Press the hip and knee bearings** (40 min) — Heat-set inserts, then 608 bearings into the hip and knee hubs. Bind here and the servo fights the bearing until one of them dies. _Tools: heat-set tip, vise._
4. **Build both legs as mirror pairs** (60 min) — Assemble the left leg fully, then build the right against it as a jig — mirror-image legs that differ by 2mm walk in circles forever. _Tools: M3 hex key._
5. **Assemble the pelvis and torso** (30 min) — Bolt both legs to the pelvis hub, then the torso and the pack. Put the machine in a stand or hang it from a strap before it can hold itself up. _Tools: M3 hex key, stand or strap._
6. **Two buses, three rails** (60 min) — Legs on bus 0, arms and head on bus 1, each bus on its own UBEC from its own pack. One shared rail will brown out the MCU the first time a hip stalls. _Tools: soldering iron, multimeter._
7. **Address every servo before assembly** (30 min) — Connect one servo at a time and set its ID while it is still off the robot. Renumbering a servo inside an assembled leg is an hour of disassembly. _Tools: bus adapter, PC._
8. **First power: standing pose, low torque** (25 min) — Run the generated stand pose with torque limits at 30%. Watch joint loads on the serial log — a joint that climbs and stays high is binding. _Tools: USB-C, bench supply._
9. **Bring up the MCU** (15 min) — Flash the generated sketch. Open the serial monitor at 115200 and confirm you see the boot line before you send any command. _Tools: USB-C cable, arduino-cli._
10. **Load test** (20 min) — Run the machine unloaded for 5 minutes, then measure current at the pack: typical should be ~10.6A, and no connector should be more than hand-warm. _Tools: multimeter / clamp meter._
11. **Verify limits** (15 min) — Walk every joint to its soft limit slowly and confirm the firmware stops it. All sizing checks passed at build time.

## Design notes

- Sized at 550mm standing with 4079g all-up: ankle 2.37 N·m, knee 2.33, hip 1.38, shoulder 1.79 N·m (with a 1.8× factor).
- Two serial buses, not 23 PWM channels: legs on bus 0, arms and head on bus 1. Half-duplex TTL, 1Mbps, each servo addressable — and every servo reports position and load back, which a biped cannot work without.
- Bracket the hips and knees with 608 bearings and print those parts in PETG, not PLA: a humanoid takes repeated shock loads at exactly those two joints.
